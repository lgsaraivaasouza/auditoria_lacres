Tabela principal de Controle de Lacres CREATE TABLE IF NOT EXISTS public.controle_lacres ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, data_envio TIMESTAMPTZ NOT NULL DEFAULT NOW(), data_referencia DATE NOT NULL DEFAULT (NOW() AT TIME ZONE 'America/Sao_Paulo')::DATE, motorista TEXT NOT NULL, responsavel_cd TEXT NOT NULL, loja_destino TEXT NOT NULL, numero_lacre TEXT NOT NULL, status TEXT NOT NULL DEFAULT 'Enviado' CHECK (status IN ('Enviado', 'Recebido')), responsavel_loja TEXT, assinatura_url TEXT, data_recebimento TIMESTAMPTZ ); CREATE INDEX IF NOT EXISTS idx_lacres_status_data ON public.controle_lacres (status, data_referencia DESC); ALTER TABLE public.controle_lacres ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS "Acesso completo controle_lacres" ON public.controle_lacres; CREATE POLICY "Acesso completo controle_lacres" ON public.controle_lacres FOR ALL USING (true) WITH CHECK (true); GRANT SELECT, INSERT, UPDATE, DELETE ON public.controle_lacres TO anon, authenticated; GRANT USAGE, SELECT ON SEQUENCE public.controle_lacres_id_seq TO anon, authenticated; -- 2. Tabela de Motoristas e Responsáveis CD (editável na aba Gestão) CREATE TABLE IF NOT EXISTS public.equipe_lacres ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, tipo TEXT NOT NULL CHECK (tipo IN ('motorista', 'responsavel_cd')), nome TEXT NOT NULL, ativo BOOLEAN NOT NULL DEFAULT true, criado_em TIMESTAMPTZ DEFAULT NOW(), UNIQUE (tipo, nome) ); INSERT INTO public.equipe_lacres (tipo, nome, ativo) VALUES ('motorista', 'Donato', true), ('motorista', 'Clayton', true), ('responsavel_cd', 'Ariadine C. Monte', true), ('responsavel_cd', 'Caio F. Santos', true), ('responsavel_cd', 'Luís Gustavo Saraiva', true) ON CONFLICT (tipo, nome) DO NOTHING; ALTER TABLE public.equipe_lacres ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS "Leitura equipe_lacres" ON public.equipe_lacres; CREATE POLICY "Leitura equipe_lacres" ON public.equipe_lacres FOR SELECT USING (true); GRANT SELECT ON public.equipe_lacres TO anon, authenticated; -- 3. Políticas do seu bucket "assinaturas" DROP POLICY IF EXISTS "Upload bucket assinaturas" ON storage.objects; CREATE POLICY "Upload bucket assinaturas" ON storage.objects FOR INSERT WITH CHECK (bucket_id = 'assinaturas'); DROP POLICY IF EXISTS "Leitura bucket assinaturas" ON storage.objects; CREATE POLICY "Leitura bucket assinaturas" ON storage.objects FOR SELECT USING (bucket_id = 'assinaturas'); -- 4. RPC: Confirmar recebimento na loja com assinatura única CREATE OR REPLACE FUNCTION public.confirmar_recebimento_lacre( p_ids BIGINT[], p_responsavel_loja TEXT, p_assinatura_url TEXT ) RETURNS JSONB LANGUAGE plpgsql SECURITY DEFINER AS $$ DECLARE v_atualizados INTEGER; BEGIN IF TRIM(COALESCE(p_responsavel_loja, '')) = '' THEN RETURN jsonb_build_object('ok', false, 'erro', 'Informe o nome do responsável pela auditoria na loja.'); END IF; IF TRIM(COALESCE(p_assinatura_url, '')) = '' THEN RETURN jsonb_build_object('ok', false, 'erro', 'A assinatura na tela é obrigatória.'); END IF; UPDATE public.controle_lacres SET status = 'Recebido', responsavel_loja = TRIM(p_responsavel_loja), assinatura_url = p_assinatura_url, data_recebimento = NOW() WHERE id = ANY(p_ids) AND status = 'Enviado'; GET DIAGNOSTICS v_atualizados = ROW_COUNT; IF v_atualizados = 0 THEN RETURN jsonb_build_object('ok', false, 'erro', 'Nenhum lacre pendente encontrado para confirmar.'); END IF; RETURN jsonb_build_object('ok', true, 'atualizados', v_atualizados); END; $$; -- 5. RPC: Validação de PIN (Equipe: 2026 | Gestor: 1993) CREATE OR REPLACE FUNCTION public.validar_pin_lacres( p_pin TEXT, p_tipo TEXT ) RETURNS JSONB LANGUAGE plpgsql SECURITY DEFINER AS $$ BEGIN IF p_tipo = 'equipe' AND TRIM(COALESCE(p_pin, '')) = '2026' THEN RETURN jsonb_build_object('ok', true); ELSIF p_tipo = 'gestor' AND TRIM(COALESCE(p_pin, '')) = '1993' THEN RETURN jsonb_build_object('ok', true); END IF; RETURN jsonb_build_object('ok', false, 'erro', 'PIN incorreto. Acesso negado.'); END; $$; -- 6. RPC: Gerenciar Motoristas e Responsáveis CD CREATE OR REPLACE FUNCTION public.gerenciar_equipe_lacres( p_pin TEXT, p_acao TEXT, p_tipo TEXT DEFAULT NULL, p_nome TEXT DEFAULT NULL, p_id BIGINT DEFAULT NULL ) RETURNS JSONB LANGUAGE plpgsql SECURITY DEFINER AS $$ BEGIN IF TRIM(COALESCE(p_pin, '')) <> '1993' THEN RETURN jsonb_build_object('ok', false, 'erro', 'PIN do Gestor incorreto.'); END IF; IF p_acao = 'listar' THEN RETURN jsonb_build_object( 'ok', true, 'equipe', COALESCE(( SELECT jsonb_agg(jsonb_build_object( 'id', id, 'tipo', tipo, 'nome', nome, 'ativo', ativo ) ORDER BY tipo ASC, nome ASC) FROM public.equipe_lacres ), '[]'::JSONB) ); ELSIF p_acao = 'adicionar' THEN INSERT INTO public.equipe_lacres (tipo, nome, ativo) VALUES (p_tipo, TRIM(p_nome), true) ON CONFLICT (tipo, nome) DO UPDATE SET ativo = true; RETURN jsonb_build_object('ok', true); ELSIF p_acao = 'toggle' THEN UPDATE public.equipe_lacres SET ativo = NOT ativo WHERE id = p_id; RETURN jsonb_build_object('ok', true); END IF; RETURN jsonb_build_object('ok', false, 'erro', 'Ação inválida.'); END; $$; GRANT EXECUTE ON FUNCTION public.confirmar_recebimento_lacre(BIGINT[], TEXT, TEXT) TO anon, authenticated; GRANT EXECUTE ON FUNCTION public.validar_pin_lacres(TEXT, TEXT) TO anon, authenticated; GRANT EXECUTE ON FUNCTION public.gerenciar_equipe_lacres(TEXT, TEXT, TEXT, TEXT, BIGINT) TO anon, authenticated; -- 7. Ativa o Realtime com segurança (não dá erro se já estiver ativo) DO $$ BEGIN IF NOT EXISTS ( SELECT 1 FROM pg_publication_tables WHERE pubname = 'supabase_realtime' AND schemaname = 'public' AND tablename = 'controle_lacres' ) THEN ALTER PUBLICATION supabase_realtime ADD TABLE public.controle_lacres; END IF; END $$;
